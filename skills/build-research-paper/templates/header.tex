% Pandoc preamble for markdown → LaTeX → PDF builds.
% Included via `pandoc --include-in-header=header.tex`.
% This is the "escape hatch" workflow; the default is hand-authored main.tex.

% ── Page geometry ────────────────────────────────────────────────────────────
\usepackage[letterpaper,margin=1in,top=1in,bottom=1in]{geometry}

% ── Typography ───────────────────────────────────────────────────────────────
\usepackage{microtype}
\setlength{\emergencystretch}{3em}
\sloppy

% ── Tables ───────────────────────────────────────────────────────────────────
\usepackage{etoolbox}
\usepackage{longtable}
\usepackage{array}
\usepackage{booktabs}
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.12}
\AtBeginEnvironment{longtable}{\footnotesize}
\AtBeginEnvironment{tabular}{\footnotesize}
\setlength{\LTleft}{0pt}
\setlength{\LTright}{0pt}

% ── Headings ─────────────────────────────────────────────────────────────────
\usepackage{titlesec}
\usepackage{xcolor}
\definecolor{accentblue}{HTML}{1B5FA8}
\titleformat*{\section}{\Large\bfseries\color{black!85}}
\titleformat*{\subsection}{\large\bfseries\color{black!80}}
\titleformat*{\subsubsection}{\normalsize\bfseries\color{black!75}}

% ── Figures pinned in place ──────────────────────────────────────────────────
\usepackage{float}
\makeatletter
\def\fps@figure{H}
\makeatother

% ── Math ─────────────────────────────────────────────────────────────────────
\usepackage{amsmath,amssymb,amsthm}
\usepackage{mathtools}

% ── Hyperlinks (load last) ───────────────────────────────────────────────────
\usepackage[hidelinks]{hyperref}
